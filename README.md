# Домашнее задание к лабораторной работе 2
## Условие задачи
Два поезда длиной L1 и L2 движутся со скоростями V1 и V2 навстречу друг другу. Определить продолжительность их прохождения один против другого.

## 1. Алгоритм и блок-схема
### Алгоритм
1. Начало

2. Объявить переменные:

а) L1 (200.0) - длина первого поезда (м)
б) L2 (300.0) - длина второго поезда (м)
в) V1 (15.0) - скорость первого поезда (м/с)
г) V2 (20.0) - скорость второго поезда (м/с)

3. Вычислить суммарную длину поездов:
lengthsum = L1 + L2

4. Вычислить скорость сближения (поезда идут навстречу):
speedsum = V1 + V2

5. Вычислить время прохождения поездов друг против друга:
time = lengthsum / speedsum

6. Вывести все данные и результаты в тексте.

7. Конец

## 2. Реализация программы
#include <stdio.h>
#define _CRT_SECURE_NO_WARNINGS
#include <locale.h>
#include <math.h>

int main() //вариант 18
{
    setlocale(LC_ALL, "RUS");

    dz();
}

int dz()
{
    double L1 = 200.0;
    double L2 = 300.0;
    double V1 = 15.0;
    double V2 = 20.0;

    double lengthsum = L1 + L2;
    double speedsum = V1 + V2;
    double time = lengthsum / speedsum;

    printf("длина первого поезда L1 = %.0f м\n", L1);
    printf("длина второго поезда L2 = %.0f м\n", L2);
    printf("скорость первого поезда V1 = %.0f м/с\n", V1);
    printf("скорость второго поезда V2 = %.0f м/с\n", V2);
    printf("суммарная длина поездов = %.0fм\n", lengthsum);
    printf("скорость сближения = %.0fм/с\n", speedsum);
    printf("поезд длиной %.0f м со скоростью %.0f м/с и поезд длиной %.0f м со скоростью %.0f м/с проходят друг против друга за %.2f секунд.\n", L1, V1, L2, V2, time);
}

скорость второго поезда V2 = 20 м/с
суммарная длина поездов = 500м
скорость сближения = 35м/с
поезд длиной 200 м со скоростью 15 м/с и поезд длиной 300 м со скоростью 20 м/с проходят друг против друга за 14,29 секунд.
